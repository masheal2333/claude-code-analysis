"上下文满了；腾出空间"

🎯 概述

上下文压缩 是 Claude Code 的内存管理机制。当对话历史太长时，它会自动压缩以释放空间。

核心思想：

  • 💭 上下文窗口有限：Claude 的上下文窗口是有限的（如 200K tokens）
  • 📈 对话会增长：每轮对话都会增加历史记录
  • 🗜️ 智能压缩：在不丢失关键信息的前提下压缩历史

压缩系统架构

graph TB A[对话历史] --> B{Token 使用率} B -->|< 60%| C[正常模式] B -->|60-80%| D[警告模式] B -->|> 80%| E[压缩模式] E --> F[选择压缩策略] F --> G[AutoCompact] F --> H[SnipCompact] F --> I[ContextCollapse] G --> J[生成摘要] H --> K[修剪冗余] I --> L[折叠子任务] J --> M[压缩后历史] K --> M L --> M M --> N[继续对话]

📝 AutoCompact (自动摘要) 详解

🎯 核心原理

使用 LLM 生成对话摘要，保留关键信息，压缩详细内容。

🔄 工作流程

graph LR A[对话历史] --> B[检测 Token 使用] B --> C{超过阈值?} C -->|否| D[继续对话] C -->|是| E[提取待压缩消息] E --> F[调用 LLM 生成摘要] F --> G[保留系统消息] G --> H[替换为摘要] H --> I[继续对话]

// services/compact/autoCompact.ts

interface AutoCompactConfig {
  threshold: number;        // Token 阈值（默认 0.8）
  maxSummaryTokens: number; // 摘要最大 Token 数
  preserveSystem: boolean;  // 是否保留系统消息
}

async function autoCompact(
  messages: Message[],
  config: AutoCompactConfig
): Promise {
  // 1. 计算当前 Token 使用量
  const tokenCount = countTokens(messages);
  const maxTokens = getModelContextLimit();
  
  // 2. 检查是否需要压缩
  if (tokenCount / maxTokens < config.threshold) {
    return messages; // 不需要压缩
  }
  
  // 3. 提取待压缩的消息（排除系统消息）
  const toCompress = messages.filter(m => m.role !== 'system');
  const systemMessages = messages.filter(m => m.role === 'system');
  
  // 4. 调用 LLM 生成摘要
  const summaryPrompt = `
    请将以下对话压缩为简洁的摘要，保留：
    1. 用户的核心需求
    2. 已完成的关键任务
    3. 重要的决策和结论
    4. 待完成的任务
    
    对话内容：
    ${toCompress.map(m => `${m.role}: ${m.content}`).join('\n')}
  `;
  
  const summary = await callLLM({
    model: 'claude-3-haiku', // 使用轻量模型生成摘要
    maxTokens: config.maxSummaryTokens,
    messages: [{ role: 'user', content: summaryPrompt }]
  });
  
  // 5. 构建压缩后的历史
  return [
    ...systemMessages,
    {
      role: 'assistant',
      content: `[对话摘要]\n${summary}\n\n[压缩时间: ${new Date().toISOString()}]`
    }
  ];
}

from dataclasses import dataclass
from typing import List, Dict, Any
from datetime import datetime

@dataclass
class AutoCompactConfig:
    threshold: float = 0.8        # Token 阈值
    max_summary_tokens: int = 2048  # 摘要最大 Token 数
    preserve_system: bool = True    # 是否保留系统消息

async def auto_compact(
    messages: List[Dict[str, Any]],
    config: AutoCompactConfig
) -> List[Dict[str, Any]]:
    # 1. 计算当前 Token 使用量
    token_count = count_tokens(messages)
    max_tokens = get_model_context_limit()
    
    # 2. 检查是否需要压缩
    if token_count / max_tokens < config.threshold:
        return messages  # 不需要压缩
    
    # 3. 提取待压缩的消息（排除系统消息）
    to_compress = [m for m in messages if m["role"] != "system"]
    system_messages = [m for m in messages if m["role"] == "system"]
    
    # 4. 调用 LLM 生成摘要
    dialogue = "\n".join(f'{m["role"]}: {m["content"]}' for m in to_compress)
    summary_prompt = f"""请将以下对话压缩为简洁的摘要，保留：
1. 用户的核心需求
2. 已完成的关键任务
3. 重要的决策和结论
4. 待完成的任务

对话内容：
{dialogue}"""
    
    summary = await call_llm(
        model="claude-3-haiku",
        max_tokens=config.max_summary_tokens,
        messages=[{"role": "user", "content": summary_prompt}]
    )
    
    # 5. 构建压缩后的历史
    return system_messages + [{
        "role": "assistant",
        "content": f"[对话摘要]\n{summary}\n\n[压缩时间: {datetime.utcnow().isoformat()}]"
    }]

✂️ SnipCompact (智能修剪) 详解

🎯 核心原理

删除已完成的工具调用结果，保留工具调用的摘要信息。

🔄 工作流程

graph LR A[工具调用完成] --> B[标记为可修剪] B --> C{Token 压力大?} C -->|否| D[保留完整结果] C -->|是| E[提取关键信息] E --> F[删除详细输出] F --> G[保留摘要]

💻 实现代码

// services/compact/snipCompact.ts

interface SnipableToolResult {
  toolName: string;
  originalSize: number;
  summary: string;
  canSnip: boolean;
}

async function snipCompact(
  messages: Message[]
): Promise {
  const result: Message[] = [];
  
  for (const msg of messages) {
    if (msg.role === 'tool') {
      // 1. 检查工具结果是否可以修剪
      const snipable = analyzeToolResult(msg);
      
      if (snipable.canSnip && snipable.originalSize > 1000) {
        // 2. 生成摘要
        const summary = generateToolSummary(msg);
        
        // 3. 替换为摘要
        result.push({
          role: 'tool',
          content: `[原始输出已修剪，共 ${snipable.originalSize} 字符]\n${summary}`
        });
      } else {
        result.push(msg);
      }
    } else {
      result.push(msg);
    }
  }
  
  return result;
}

function analyzeToolResult(msg: ToolMessage): SnipableToolResult {
  const content = msg.content;
  
  // 检查是否是文件读取、命令执行等可修剪的工具
  const snippableTools = ['read_file', 'execute_command', 'search'];
  const toolName = extractToolName(msg);
  
  return {
    toolName,
    originalSize: content.length,
    summary: '',
    canSnip: snippableTools.includes(toolName)
  };
}

    original_size: int
    summary: str

async def snip_compact(
    messages: List[Dict[str, Any]]
) -> List[Dict[str, Any]]:
    result = []
    
    for msg in messages:
        if msg.get("role") == "tool":
            # 1. 检查工具结果是否可以修剪
            snipable = analyze_tool_result(msg)

                # 2. 生成摘要

                # 3. 替换为摘要
                result.append({
                    "role": "tool",
                    "content": f"[原始输出已修剪，共 {snipable.original_size} 字符]\n{summary}"
                })
            else:
                result.append(msg)
        else:
            result.append(msg)
    
    return result

def analyze_tool_result(msg: Dict[str, Any]) -> SnipableToolResult:
    content = msg.get("content", "")
    
    # 检查是否是文件读取、命令执行等可修剪的工具
    snippable_tools = ["read_file", "execute_command", "search"]
    tool_name = extract_tool_name(msg)
    
    return SnipableToolResult(
        tool_name=tool_name,
        original_size=len(content),
        summary="",
        can_snip=tool_name in snippable_tools
    )

🗜️ ContextCollapse (上下文折叠) 详解

🎯 核心原理

折叠已完成的子代理任务，将详细的执行过程压缩为简洁的结果。

🔄 工作流程

graph LR A[子代理任务完成] --> B[收集执行历史] B --> C[提取关键结果] C --> D[生成折叠摘要] D --> E[替换详细历史] E --> F[保留摘要]

💻 实现代码

// services/compact/contextCollapse.ts

interface SubAgentTask {
  taskId: string;
  task: string;
  messages: Message[];
  result: any;
  status: 'completed' | 'failed';
}

async function contextCollapse(
  mainMessages: Message[],
  subAgentTasks: SubAgentTask[]
): Promise {
  const result: Message[] = [];
  
  for (const msg of mainMessages) {
    // 1. 检查是否是子代理任务的消息
    const isSubAgentMsg = subAgentTasks.some(task => 
      task.messages.some(m => m.id === msg.id)
    );
    
    if (isSubAgentMsg) {
      // 2. 找到对应的子代理任务
      const task = subAgentTasks.find(t => 
        t.messages.some(m => m.id === msg.id)
      );
      
      // 3. 只保留第一条和最后一条消息（任务描述和结果）
      const isFirstMsg = task.messages[0].id === msg.id;
      const isLastMsg = task.messages[task.messages.length - 1].id === msg.id;
      
      if (isFirstMsg) {
        result.push(msg); // 保留任务描述
      } else if (isLastMsg) {
        // 替换为折叠摘要
        result.push({
          role: 'assistant',
          content: `[子代理任务完成]\n任务: ${task.task}\n状态: ${task.status}\n结果: ${JSON.stringify(task.result).substring(0, 200)}...`
        });
      }
      // 中间的详细消息被丢弃
    } else {
      result.push(msg);
    }
  }
  
  return result;
}

from dataclasses import dataclass
from typing import List, Dict, Any
import json

@dataclass
class SubAgentTask:
    task_id: str
    task: str
    messages: List[Dict[str, Any]]
    result: Any
    status: str  # "completed" | "failed"

async def context_collapse(
    main_messages: List[Dict[str, Any]],
    sub_agent_tasks: List[SubAgentTask]
) -> List[Dict[str, Any]]:
    result = []
    
    # 建立消息 ID 到任务的映射
    task_map = {}
    for task in sub_agent_tasks:
        for msg in task.messages:
            task_map[msg["id"]] = task
    
    for msg in main_messages:
        # 1. 检查是否是子代理任务的消息
        task = task_map.get(msg["id"])
        
        if task is not None:
            # 2. 只保留第一条和最后一条消息
            is_first = task.messages[0]["id"] == msg["id"]
            is_last = task.messages[-1]["id"] == msg["id"]
            
            if is_first:
                result.append(msg)  # 保留任务描述
            elif is_last:
                result.append({
                    "role": "assistant",
                    "content": (
                        f"[子代理任务完成]\n"
                        f"任务: {task.task}\n"
                        f"状态: {task.status}\n"
                        f"结果: {json.dumps(task.result)[:200]}..."
                    )
                })
            # 中间的详细消息被丢弃
        else:
            result.append(msg)
    
    return result

// services/compact/index.ts

// 1. 压缩策略接口
interface CompressionStrategy {
  name: string;
  priority: number;  // 优先级，数字越小优先级越高
  shouldCompress: (context: CompressionContext) => boolean;
  compress: (messages: Message[]) => Promise;
}

// 2. 压缩上下文
interface CompressionContext {
  tokenCount: number;
  maxTokens: number;
  messageCount: number;
  hasSubAgentTasks: boolean;
  hasToolResults: boolean;
}

// 3. 压缩管理器
class CompressionManager {
  private strategies: CompressionStrategy[] = [];
  
  registerStrategy(strategy: CompressionStrategy) {
    this.strategies.push(strategy);
    this.strategies.sort((a, b) => a.priority - b.priority);
  }
  
  async compress(messages: Message[]): Promise {
    const context = this.buildContext(messages);
    
    // 按优先级尝试压缩策略
    for (const strategy of this.strategies) {
      if (strategy.shouldCompress(context)) {
        console.log(`使用压缩策略: ${strategy.name}`);
        return await strategy.compress(messages);
      }
    }
    
    return messages; // 没有策略触发
  }
  
  private buildContext(messages: Message[]): CompressionContext {
    return {
      tokenCount: countTokens(messages),
      maxTokens: getModelContextLimit(),
      messageCount: messages.length,
      hasSubAgentTasks: messages.some(m => m.role === 'tool' && m.content.includes('subagent')),
      hasToolResults: messages.some(m => m.role === 'tool')
    };
  }
}

@dataclass
class CompressionContext:
    token_count: int
    max_tokens: int
    message_count: int
    has_sub_agent_tasks: bool
    has_tool_results: bool

# 3. 压缩管理器
class CompressionManager:
    def __init__(self):
        self.strategies: List[CompressionStrategy] = []
    
    def register_strategy(self, strategy: CompressionStrategy):
        self.strategies.append(strategy)
        self.strategies.sort(key=lambda s: s.priority)
    
    async def compress(self, messages: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        context = self._build_context(messages)
        
        # 按优先级尝试压缩策略
        for strategy in self.strategies:
            if strategy.should_compress(context):
                print(f"使用压缩策略: {strategy.name}")
                return await strategy.compress(messages)
        
        return messages  # 没有策略触发

            token_count=count_tokens(messages),
            max_tokens=get_model_context_limit(),
            message_count=len(messages),
            has_sub_agent_tasks=any(
                m.get("role") == "tool" and "subagent" in str(m.get("content", ""))
                for m in messages
            ),
            has_tool_results=any(m.get("role") == "tool" for m in messages)
        )
