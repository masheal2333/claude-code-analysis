"大目标 → 小任务 → 磁盘"

← 返回首页 概述 代码分析 任务格式 工作流程

🎯 概述

持久化任务系统 是 Claude Code 的任务管理机制。它将大目标分解为小任务，并持久化到磁盘，支持状态跟踪和恢复。

核心特性：

  • 📊 任务图：支持任务之间的依赖关系
  • 💾 持久化：任务状态保存到磁盘，可跨会话恢复
  • 🔄 状态跟踪：pending → in_progress → completed
  • 📝 详细记录：每个任务都有详细的描述和进度

interface Task {
    id: string;
    content: string;
    status: 'pending' | 'in_progress' | 'completed' | 'cancelled';
    priority: 'high' | 'medium' | 'low';
    dependencies: string[];
    createdAt: Date;
    updatedAt: Date;
}

async function createTask(params: CreateTaskParams): Promise<Task> {
    const task: Task = {
        id: generateId(),
        content: params.content,

        dependencies: params.dependencies || [],
        createdAt: new Date(),

from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import List, Optional
import uuid

class TaskStatus(Enum):
    PENDING = "pending"
    IN_PROGRESS = "in_progress"
    COMPLETED = "completed"
    CANCELLED = "cancelled"

class TaskPriority(Enum):
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"

@dataclass
class Task:
    id: str
    content: str
    status: TaskStatus
    priority: TaskPriority
    dependencies: List[str]
    created_at: datetime
    updated_at: datetime

def create_task(
    content: str,
    priority: TaskPriority = TaskPriority.MEDIUM,
    dependencies: Optional[List[str]] = None
) -> Task:
    task = Task(
        id=str(uuid.uuid4()),
        content=content,
        status=TaskStatus.PENDING,
        priority=priority,
        dependencies=dependencies or [],
        created_at=datetime.now(),
        updated_at=datetime.now()
    )
    
    # 保存到磁盘
    save_task_to_disk(task)
    
    return task

async function updateTask(
    taskId: string, 
    updates: Partial<Task>
): Promise<Task> {
    const task = await getTaskFromDisk(taskId);
    
    // 更新字段
    Object.assign(task, updates, { updatedAt: new Date() });
    
    // 验证状态转换
    validateStatusTransition(task.status, updates.status);
    
    // 保存到磁盘
    await saveTaskToDisk(task);
    
    return task;
}
from typing import Dict, Any

def update_task(task_id: str, updates: Dict[str, Any]) -> Task:
    task = get_task_from_disk(task_id)
    
    # 更新字段
    for key, value in updates.items():
        if hasattr(task, key):
            setattr(task, key, value)
    task.updated_at = datetime.now()
    
    # 验证状态转换
    if "status" in updates:
        validate_status_transition(task.status, updates["status"])
    
    # 保存到磁盘
    save_task_to_disk(task)
    
    return task
