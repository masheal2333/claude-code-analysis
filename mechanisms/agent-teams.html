🎯 概述

代理团队系统 允许创建持久的代理团队，它们可以协作完成复杂任务。

核心特性：

  • 👥 持久团队：团队成员长期存在
  • 📬 异步通信：通过邮箱系统通信
  • 🎯 任务分配：智能分配任务给合适的成员
  • 🔄 协作完成：团队成员协作完成任务

TeamCreate - 创建团队

interface TeamCreateParams {
  name: string;
  members: TeamMember[];
  communicationProtocol?: 'async' | 'sync';
}

async function createTeam(params: TeamCreateParams) {
  // 1. 创建团队
  const team = {
    id: generateTeamId(),
    name: params.name,
    members: params.members,
    createdAt: new Date()
  };
  
  // 2. 为每个成员创建邮箱
  for (const member of params.members) {
    await createMailbox(member.id);
  }
  
  // 3. 建立通信通道
  await setupCommunicationChannels(team);
  
  return team;
}
from dataclasses import dataclass, field
from typing import List, Optional
from datetime import datetime

@dataclass
class TeamMember:
    id: str
    name: str
    role: str

@dataclass
class TeamCreateParams:
    name: str
    members: List[TeamMember]
    communication_protocol: Optional[str] = "async"  # "async" | "sync"

async def create_team(params: TeamCreateParams) -> dict:
    # 1. 创建团队
    team = {
        "id": generate_team_id(),
        "name": params.name,
        "members": params.members,
        "created_at": datetime.utcnow().isoformat()
    }
    
    # 2. 为每个成员创建邮箱
    for member in params.members:
        await create_mailbox(member.id)
    
    # 3. 建立通信通道
    await setup_communication_channels(team)
    
    return team
